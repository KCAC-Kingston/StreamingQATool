cmake_minimum_required(VERSION 3.28)
project(StreamingQAToolTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets Network)
add_executable(live-control-tests live-control-tests.cpp ../src/live-control.cpp ../src/service-manager-client.cpp ../src/settings.cpp)
target_sources(live-control-tests PRIVATE ../src/live-control-requests.cpp)
target_include_directories(live-control-tests PRIVATE ../src)
target_link_libraries(live-control-tests PRIVATE Qt6::Widgets Qt6::Network)
if(WIN32)
  target_sources(live-control-tests PRIVATE ../src/windows-http.cpp)
  target_link_libraries(live-control-tests PRIVATE winhttp)
endif()
enable_testing()
add_test(NAME live-control COMMAND live-control-tests)
set_tests_properties(live-control PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=minimal")

# Opt-in live read-only test; deliberately not registered with CTest.
add_executable(connection-check connection-check.cpp ../src/service-manager-client.cpp)
target_include_directories(connection-check PRIVATE ../src)
target_link_libraries(connection-check PRIVATE Qt6::Core Qt6::Network)
if(WIN32)
  target_sources(connection-check PRIVATE ../src/windows-http.cpp)
  target_link_libraries(connection-check PRIVATE winhttp)
endif()

add_executable(event-log-tests event-log-tests.cpp ../src/event-format.cpp ../src/event-log.cpp ../src/local-websocket.cpp ../src/openlp-monitor.cpp ../src/companion-monitor.cpp)
set_target_properties(event-log-tests PROPERTIES AUTOMOC ON)
target_include_directories(event-log-tests PRIVATE ../src)
target_link_libraries(event-log-tests PRIVATE Qt6::Core Qt6::Network)
add_test(NAME event-log COMMAND event-log-tests)
