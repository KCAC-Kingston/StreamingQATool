cmake_minimum_required(VERSION 3.28)
project(StreamingQAToolTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Widgets Network)
add_executable(live-control-tests live-control-tests.cpp ../src/live-control.cpp ../src/service-manager-client.cpp ../src/settings.cpp)
target_sources(live-control-tests PRIVATE ../src/live-control-requests.cpp)
target_include_directories(live-control-tests PRIVATE ../src)
target_link_libraries(live-control-tests PRIVATE Qt6::Widgets Qt6::Network)
if(WIN32)
  target_sources(live-control-tests PRIVATE ../src/windows-http.cpp)
  target_link_libraries(live-control-tests PRIVATE winhttp)
endif()
enable_testing()
add_test(NAME live-control COMMAND live-control-tests)
set_tests_properties(live-control PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=minimal")
